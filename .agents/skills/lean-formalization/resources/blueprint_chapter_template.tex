\begin{theorem}\label{thm:my_theorem}
\lean{FracLap.my_theorem}
\uses{def:my_definition}
\leanok
If $x$ satisfies Definition~\ref{def:my_definition}, then $x \ge 0$.
\end{theorem}

\begin{proof}
\uses{def:my_definition}
% Add \leanok here only once FracLap.my_theorem has zero `sorry`s!
