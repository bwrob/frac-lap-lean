\chapter{Yosida Approximations and Semigroup Generation}

This chapter mirrors \texttt{FracLap/Core/HilleYosidaGeneration.lean}, onboarding Banach-algebra exponential bounds, Duhamel variation-of-constants estimates, bounded Yosida approximants $G_\alpha = \alpha(\alpha R_\alpha - I)$, and the construction of the strongly continuous contraction semigroup generated by a contractive resolvent.

\begin{lemma}\label{lem:banach_exp_contraction}
\lean{FracLap.norm_exp_le_exp_norm_of_norm_one_le, FracLap.norm_exp_le_exp_norm, FracLap.norm_exp_smul_sub_one_le_one_of_norm_one_le, FracLap.norm_exp_smul_sub_one_le_one, FracLap.norm_exp_continuousLinearMap_smul_sub_id_le_one, FracLap.hasSum_exp_smul_sub_one}
\leanok
In a real Banach algebra $A$ with $\lVert 1 \rVert \le 1$, we have $\lVert \exp(a) \rVert \le e^{\lVert a \rVert}$ for all $a \in A$. Whenever $c \ge 0$ and $Q \in A$ satisfies $\lVert Q \rVert \le 1$, the exponential $\exp\bigl(c(Q - 1)\bigr)$ is a contraction ($\lVert \exp(c(Q - 1)) \rVert \le 1$); in particular, this holds on the operator algebra $\mathcal{B}(E)$ even when $E = \{0\}$.
\end{lemma}

\begin{proof}
\leanok
Bounding each term $(n!)^{-1} \lVert a^n \rVert \le (n!)^{-1} \lVert a \rVert^n$ in the exponential power series gives $\lVert \exp(a) \rVert \le e^{\lVert a \rVert}$. Since $cQ$ commutes with $-c \cdot 1$, we obtain $\exp\bigl(c(Q - 1)\bigr) = \exp(cQ) \exp(-c \cdot 1)$ with norm bounded by $e^{c \lVert Q \rVert} e^{-c} \le e^c e^{-c} = 1$.
\end{proof}

\begin{lemma}\label{lem:duhamel_estimate}
\lean{FracLap.exp_sub_exp_apply_eq_integral, FracLap.norm_exp_sub_exp_apply_le, FracLap.norm_exp_apply_sub_le}
\leanok
Let $A, B \in \mathcal{B}(E)$ be commuting bounded linear operators on a real Banach space $E$. Then for every $t \in \mathbb{R}$ and $x \in E$,
\[
  e^{tB} x - e^{tA} x = \int_0^t e^{(t - s)A} e^{sB} (B - A) x\, \mathrm{d}s.
\]
Consequently, if $t \ge 0$ and $\lVert e^{sA} \rVert \le 1$ and $\lVert e^{sB} \rVert \le 1$ for all $s \in [0, t]$, then
\[
  \lVert e^{tB} x - e^{tA} x \rVert \le t \lVert (B - A) x \rVert, \qquad \lVert e^{tB} x - x \rVert \le t \lVert B x \rVert.
\]
\end{lemma}

\begin{proof}
\leanok
Differentiating $s \mapsto e^{(t - s)A} e^{sB} x$ using the product rule and commutation of $A$ with $e^{sB}$ yields $e^{(t - s)A} e^{sB}(B - A)x$, so the fundamental theorem of calculus gives Duhamel's identity. Taking norms under the integral against unit operator-norm bounds gives both estimates.
\end{proof}

\begin{definition}\label{def:yosida_approximations}
\lean{FracLap.naturalShift, FracLap.naturalShift_coe, FracLap.tendsto_naturalShift_atTop, FracLap.ContractiveResolvent.yosidaGenerator, FracLap.ContractiveResolvent.yosidaOperator, FracLap.ContractiveResolvent.yosidaOperator_zero, FracLap.ContractiveResolvent.yosidaOperator_add, FracLap.ContractiveResolvent.norm_yosidaOperator_le_one, FracLap.ContractiveResolvent.hasSum_yosidaOperator_apply, FracLap.ContractiveResolvent.continuous_yosidaOperator_apply, FracLap.ContractiveResolvent.yosidaSemigroup, FracLap.ContractiveResolvent.yosidaSemigroup_apply}
\uses{def:contractive_resolvent, lem:banach_exp_contraction, def:c0_contraction_semigroup}
\leanok
For a contractive resolvent $(R_\alpha)_{\alpha > 0}$ on a real Banach space $E$, the \emph{Yosida generator} at shift $\alpha > 0$ is the bounded operator
\[
  G_\alpha = \alpha(\alpha R_\alpha - I) \in \mathcal{B}(E),
\]
and its exponential semigroup $S_\alpha(t) = \exp(t G_\alpha)$ ($t \ge 0$) is a strongly continuous contraction semigroup on $E$. We write $\alpha_n = n + 1$ for the canonical sequence of positive shifts tending to $+\infty$.
\end{definition}

\begin{lemma}\label{lem:yosida_commutativity_and_core_limit}
\lean{FracLap.ContractiveResolvent.yosidaGenerator_commute, FracLap.ContractiveResolvent.yosidaOperator_commute, FracLap.ContractiveResolvent.yosidaGenerator_apply_operator, FracLap.ContractiveResolvent.tendsto_yosidaGenerator_apply_operator, FracLap.ContractiveResolvent.tendsto_yosidaGenerator_naturalShift_apply_operator, FracLap.ContractiveResolvent.cauchySeq_yosidaGenerator_naturalShift_apply_operator}
\uses{def:yosida_approximations, thm:contractive_resolvent_normalization}
\leanok
The Yosida generators $G_\alpha$ and their exponentials $S_\alpha(t)$ mutually commute across all shifts and times. Moreover, on the range of any fixed resolvent $R_\mu$,
\[
  G_\alpha(R_\mu y) = \alpha R_\alpha(\mu R_\mu y - y) \xrightarrow{\alpha \to \infty} \mu R_\mu y - y.
\]
In particular, $\bigl(G_{\alpha_n}(R_\mu y)\bigr)_n$ is Cauchy in $E$.
\end{lemma}

\begin{proof}
\uses{def:yosida_approximations, thm:contractive_resolvent_normalization}
\leanok
Commutativity of $G_\alpha$ and $G_\beta$ follows directly from $R_\alpha R_\beta = R_\beta R_\alpha$, and carries over to exponentials. Multiplying the identity of Theorem~\ref{thm:contractive_resolvent_normalization} by $\alpha$ gives $G_\alpha(R_\mu y) = \alpha R_\alpha(\mu R_\mu y - y)$, which converges to $\mu R_\mu y - y$ as $\alpha \to \infty$.
\end{proof}

\begin{theorem}\label{thm:yosida_strong_limit_and_generation}
\lean{FracLap.ContractiveResolvent.cauchySeq_yosidaOperator_naturalShift_apply_operator, FracLap.ContractiveResolvent.yosidaOperatorSequence, FracLap.ContractiveResolvent.norm_yosidaOperatorSequence_le_one, FracLap.ContractiveResolvent.cauchySeq_yosidaOperatorSequence_apply, FracLap.ContractiveResolvent.yosidaStrongLimit, FracLap.ContractiveResolvent.tendsto_yosidaOperator_naturalShift_apply, FracLap.ContractiveResolvent.norm_yosidaStrongLimit_le_one, FracLap.ContractiveResolvent.yosidaStrongLimit_zero, FracLap.ContractiveResolvent.yosidaStrongLimit_add, FracLap.ContractiveResolvent.exists_norm_yosidaStrongLimit_apply_sub_on_range_le, FracLap.ContractiveResolvent.tendsto_yosidaStrongLimit_apply_operator_zero, FracLap.ContractiveResolvent.tendsto_yosidaStrongLimit_apply_zero, FracLap.ContractiveResolvent.generatedSemigroup, FracLap.ContractiveResolvent.generatedSemigroup_apply, FracLap.ContractiveResolvent.tendsto_yosidaSemigroup_naturalShift_apply_generatedSemigroup}
\uses{lem:duhamel_estimate, lem:yosida_commutativity_and_core_limit, def:strong_operator_limit, thm:c0_orbit_continuity_from_zero}
\leanok
Let $(R_\alpha)_{\alpha > 0}$ be a contractive resolvent on a real Banach space $E$. For every $t \ge 0$ and $x \in E$, the canonical Yosida exponential sequence $S_{\alpha_n}(t)x = \exp(t G_{\alpha_n})x$ converges in $E$ to a contraction operator $S(t)x$. The resulting family $\bigl(S(t)\bigr)_{t \ge 0}$ satisfies $S(0) = I$, $S(s + t) = S(s) \circ S(t)$, $\lVert S(t) \rVert \le 1$, and $\lim_{t \to 0^+} S(t)x = x$ for every $x \in E$, hence forms a strongly continuous contraction semigroup on $E$.
\end{theorem}

\begin{proof}
\uses{lem:duhamel_estimate, lem:yosida_commutativity_and_core_limit, def:strong_operator_limit, thm:c0_orbit_continuity_from_zero}
\leanok
For $x = R_\mu y$, Duhamel's estimate (Lemma~\ref{lem:duhamel_estimate}) gives
\[
  \lVert S_{\alpha_m}(t)x - S_{\alpha_n}(t)x \rVert \le t \lVert G_{\alpha_m} x - G_{\alpha_n} x \rVert,
\]
which shows that $(S_{\alpha_n}(t)x)_n$ is Cauchy on the dense range of $R_1$, hence on all of $E$ by Lemma~\ref{lem:dense_core_convergence}. Its strong operator limit $S(t)$ inherits contractivity, $S(0) = I$, and the semigroup law. On $R_1(E)$, uniform boundedness of $\lVert G_{\alpha_n}(R_1 y) \rVert \le C$ and Lemma~\ref{lem:duhamel_estimate} imply $\lVert S(t)(R_1 y) - R_1 y \rVert \le C t \to 0$ as $t \to 0^+$. Extending strong continuity at $t = 0$ by density (Lemma~\ref{lem:dense_core_convergence}) and upgrading via Theorem~\ref{thm:c0_orbit_continuity_from_zero} constructs the generated semigroup $S$.
\end{proof}
